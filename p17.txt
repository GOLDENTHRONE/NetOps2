WHY THIS EXISTS (one paragraph)
--------------------------------------------------------------------------------
The existing "health" column in the Applications tab is misleading. Full
technical review is in health.txt on this same branch. Short version: only
Deployment-family objects can produce non-'success' status, Pod is not even
fetched, and 14 other kinds always return 'success'. "Unhealthy" is
effectively dead code, and real failures (failed Jobs, Pending PVCs, empty
Endpoints, quota exhaustion, bare-pod CrashLoopBackOff) all show as
"Healthy". p17 fixes this fork-locally with the smallest possible touch and
with zero change in what other parts of Headlamp (Resources column,
ProjectDetails, ResourceMap, backend, plugins) render or fetch.

Approach: comment out (do NOT delete) the upstream "health" column, add a
new "Status" column driven by a new local helper `getLocalHealth`, and
fetch Pods LOCALLY inside the new column's Cell — never touching the shared
`defaultApiResources` list, never registering Pod globally via any plugin,
never mutating Redux `state.projects.apiResources`. As a result, the extra
Pod data exists only in-memory of that one Cell; every other consumer of
shared project data behaves exactly as before.

     import { countBy } from 'lodash';
     import type { KubeObject } from '../../lib/k8s/KubeObject';
     import { getReadyReplicas, getTotalReplicas } from '../../lib/util';

     export type LocalHealthStatus = 'success' | 'warning' | 'error';
     export type LocalHealthBadge  = LocalHealthStatus | 'empty';

     export interface LocalHealthResult {
       status: LocalHealthBadge;   // 'empty' when items.length === 0
       label: 'Healthy' | 'Degraded' | 'Unhealthy' | 'No Resources';
       rank: 0 | 1 | 2 | 3;        // 0 empty, 1 healthy, 2 degraded, 3 unhealthy
       icon: string;               // mdi:* icon id
       reasons: string[];          // ≤ 10 items; suffix "...and N more" if capped
     }

     const REASON_CAP = 10;

     function localGetItemStatus(o: KubeObject): {
       s: LocalHealthStatus;
       reason?: string;
     } {
       const kind = o.kind;
       const now  = Date.now();
       const ageMs = (ts?: string) => (ts ? now - new Date(ts).getTime() : 0);
       const olderThan = (ts: string | undefined, ms: number) => ageMs(ts) > ms;

       if (kind === 'Pod') {
         const s: any = (o as any).status ?? {};
         const phase: string | undefined = s.phase;
         const conds: any[] = s.conditions ?? [];
         const ready = conds.find(c => c.type === 'Ready')?.status === 'True';
         const waitReasons = new Set([
           'CrashLoopBackOff', 'ImagePullBackOff', 'ErrImagePull',
           'CreateContainerConfigError', 'InvalidImageName',
         ]);
         const cs: any[] = s.containerStatuses ?? [];
         const hitWait = cs.some(c => waitReasons.has(c.state?.waiting?.reason));
         if (phase === 'Failed')  return { s: 'error', reason: `Pod/${o.metadata.namespace}/${o.metadata.name}: Failed` };
         if (hitWait)             return { s: 'error', reason: `Pod/${o.metadata.namespace}/${o.metadata.name}: ${cs.find(c => waitReasons.has(c.state?.waiting?.reason))!.state.waiting.reason}` };
         if (phase === 'Running' && !ready)
                                  return { s: 'error', reason: `Pod/${o.metadata.namespace}/${o.metadata.name}: Running but NotReady` };
         if (phase === 'Pending' && olderThan(o.metadata.creationTimestamp, 5 * 60_000))
                                  return { s: 'error', reason: `Pod/${o.metadata.namespace}/${o.metadata.name}: Pending > 5m` };
         if (phase === 'Pending') return { s: 'warning', reason: `Pod/${o.metadata.namespace}/${o.metadata.name}: Pending` };
         return { s: 'success' };
       }

       if (['Deployment', 'ReplicaSet', 'StatefulSet', 'DaemonSet'].includes(kind)) {
         const spec: any = (o as any).spec ?? {};
         const desired = spec.replicas ?? getTotalReplicas(o as any);
         const ready   = getReadyReplicas(o as any) ?? 0;
         const statusReplicas: number = (o as any).status?.replicas ?? 0;
         if (desired === 0)                       return { s: 'success' };
         if (statusReplicas === 0 && desired > 0) return { s: 'error',   reason: `${kind}/${o.metadata.namespace}/${o.metadata.name}: 0/${desired} pods created` };
         if (ready < desired)                     return { s: 'warning', reason: `${kind}/${o.metadata.namespace}/${o.metadata.name}: ${ready}/${desired} ready` };
         return { s: 'success' };
       }

       if (kind === 'Job') {
         const st: any = (o as any).status ?? {};
         const sp: any = (o as any).spec ?? {};
         const failed    = st.failed    ?? 0;
         const succeeded = st.succeeded ?? 0;
         const backoff   = (sp.backoffLimit ?? 6) + 1;
         if (failed >= backoff && succeeded === 0)
           return { s: 'error', reason: `Job/${o.metadata.namespace}/${o.metadata.name}: failed (${failed}/${backoff})` };
         return { s: 'success' };
       }

       if (kind === 'CronJob') {
         const st: any = (o as any).status ?? {};
         const sp: any = (o as any).spec ?? {};
         const active = st.active?.length ?? 0;
         const cap    = sp.concurrencyPolicy === 'Forbid' ? 1 : 5;
         if (active > cap)
           return { s: 'warning', reason: `CronJob/${o.metadata.namespace}/${o.metadata.name}: ${active} active runs` };
         return { s: 'success' };
       }

       if (kind === 'PersistentVolumeClaim') {
         const phase = (o as any).status?.phase;
         if (phase === 'Lost')
           return { s: 'error',   reason: `PVC/${o.metadata.namespace}/${o.metadata.name}: Lost` };
         if (phase === 'Pending' && olderThan(o.metadata.creationTimestamp, 2 * 60_000))
           return { s: 'error',   reason: `PVC/${o.metadata.namespace}/${o.metadata.name}: Pending > 2m` };
         if (phase === 'Pending')
           return { s: 'warning', reason: `PVC/${o.metadata.namespace}/${o.metadata.name}: Pending` };
         return { s: 'success' };
       }

       if (kind === 'Endpoints') {
         const subsets = (o as any).subsets ?? [];
         const hasAddr = subsets.some((sub: any) => (sub.addresses?.length ?? 0) > 0);
         if (subsets.length === 0 || !hasAddr)
           return { s: 'warning', reason: `Endpoints/${o.metadata.namespace}/${o.metadata.name}: no endpoints` };
         return { s: 'success' };
       }

       if (kind === 'HorizontalPodAutoscaler') {
         const conds: any[] = (o as any).status?.conditions ?? [];
         const bad = conds.find(c => c.type === 'ScalingActive' && c.status === 'False' && c.reason !== 'ScalingDisabled');
         if (bad)
           return { s: 'warning', reason: `HPA/${o.metadata.namespace}/${o.metadata.name}: ${bad.reason}` };
         return { s: 'success' };
       }

       if (kind === 'Ingress') {
         const lb = (o as any).status?.loadBalancer?.ingress ?? [];
         if (lb.length === 0 && olderThan(o.metadata.creationTimestamp, 5 * 60_000))
           return { s: 'warning', reason: `Ingress/${o.metadata.namespace}/${o.metadata.name}: no address` };
         return { s: 'success' };
       }

       // ResourceQuota omitted from v1 — quantity parsing has enough edge cases
       // (Ki/Mi/m/decimal) that a wrong impl gives false positives. Add later
       // once we've picked a quantity parser.

       return { s: 'success' };
     }

     export function getLocalHealth(items: KubeObject[]): LocalHealthResult {
       if (!items || items.length === 0) {
         return { status: 'empty', label: 'No Resources', rank: 0, icon: 'mdi:help-circle', reasons: [] };
       }

       const perItem = items.map(localGetItemStatus);
       const tally = countBy(perItem, x => x.s) as Record<LocalHealthStatus, number>;
       const reasons = perItem
         .filter(x => x.s !== 'success' && x.reason)
         .map(x => x.reason!) as string[];
       const capped = reasons.slice(0, REASON_CAP);
       if (reasons.length > REASON_CAP) {
         capped.push(`…and ${reasons.length - REASON_CAP} more`);
       }

       if ((tally.error ?? 0) > 0) {
         return { status: 'error',   label: 'Unhealthy', rank: 3, icon: 'mdi:alert-circle', reasons: capped };
       }
       if ((tally.warning ?? 0) > 0) {
         return { status: 'warning', label: 'Degraded',  rank: 2, icon: 'mdi:alert',        reasons: capped };
       }
       return {   status: 'success', label: 'Healthy',   rank: 1, icon: 'mdi:check-circle', reasons: [] };
     }

     import { useMemo } from 'react';
     import { useKubeLists } from '../advancedSearch/utils/useKubeLists';
     import type { ApiResource } from '../../lib/k8s/api/v2/ApiResource';
     import type { KubeObject } from '../../lib/k8s/KubeObject';
     import { useProjectItems } from './useProjectResources';
     import type { Project } from './projectUtils';   // if a public Project type is exported;
                                                       // otherwise import from wherever
                                                       // useProjectItems' signature declares it

     const POD_RESOURCE: ApiResource = {
       apiVersion: 'v1',
       version: 'v1',
       pluralName: 'pods',
       singularName: 'pod',
       kind: 'Pod',
       isNamespaced: true,
     };

     const MAX_ITEMS       = 1000;
     const REFETCH_INTERVAL_MS = 60_000;

     export function useLocalHealthItems(project: Project) {
       // 1. shared 19-kind data — same call the Resources & health columns use
       const projectItems = useProjectItems(project, { disableWatch: true });

       // 2. LOCAL Pod fetch — scoped to this project only; NOT written to any
       //    global state; only this hook's caller ever sees these Pods.
       const localPods = useKubeLists(
         [POD_RESOURCE],
         project.clusters,
         MAX_ITEMS,
         REFETCH_INTERVAL_MS,
         project.namespaces
       );

       const items: KubeObject[] = useMemo(
         () => [...(projectItems.items ?? []), ...((localPods as any).items ?? [])],
         [projectItems.items, (localPods as any).items]
       );

       const isLoading = projectItems.isLoading || (localPods as any).isLoading;
       const errors    = [...(projectItems.errors ?? []), ...((localPods as any).errors ?? [])];
       return { items, isLoading, errors };
     }

   between Headlamp versions. Adjust the destructure to whatever fields it
   actually exposes (items, isLoading, errors are the common trio). This is
   the ONLY file where such adjustment is allowed; do not push casts into
   shared code.

HEALTH RULES v1 (already inlined in localHealth.ts above — recap)
--------------------------------------------------------------------------------
Aggregation:
  error > 0                 -> Unhealthy  (rank 3)
  warning > 0               -> Degraded   (rank 2)
  else, items.length > 0    -> Healthy    (rank 1)
  items.length === 0        -> empty ("No Resources", rank 0)

Per-kind (short-circuit at first matching rule):
  Pod             : Failed OR CrashLoopBackOff/ImagePullBackOff/etc. OR
                    Running-but-NotReady OR Pending>5min → error;
                    Pending<5min → warning; else success.
  Dep/RS/SS/DS    : desired 0 → success; created 0 with desired>0 → error;
                    ready < desired → warning; else success.
  Job             : failed >= backoffLimit+1 AND succeeded === 0 → error;
                    else success.
  CronJob         : active > cap → warning; else success.
                    (cap = 1 for Forbid, 5 otherwise)
  PVC             : Lost → error; Pending>2min → error; Pending<2min → warning;
                    else success.
  Endpoints       : no subsets or no addresses → warning; else success.
  HPA             : ScalingActive=False (not ScalingDisabled) → warning;
                    else success.
  Ingress         : no LB address AND older than 5min → warning; else success.
  Everything else : success.
