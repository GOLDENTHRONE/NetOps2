----- BEGIN HARNESS (paste as-is) ---------------------------------------------
(() => {
  if (window.__CAP) { console.log('harness already installed'); return; }
  const CAP = { perf: [], ws: [], consoleErr: {}, marks: [], installedAt: Date.now() };
  window.__CAP = CAP;

  // (1) Performance: version / SAR / clusterversions durations (same-origin => real timing)
  try {
    const po = new PerformanceObserver(list => {
      for (const e of list.getEntries()) {
        if (/\/version(\?|$)|selfsubjectrulesreviews|clusterversions/.test(e.name)) {
          const m = e.name.match(/\/clusters\/([^/]+)\//);
          CAP.perf.push({
            path: e.name.replace(/https?:\/\/[^/]+/, '').split('?')[0],
            cluster: m ? m[1] : '?',
            start: Math.round(e.startTime),
            ms: Math.round(e.duration),
          });
        }
      }
    });
    po.observe({ type: 'resource', buffered: true });
  } catch (e) { console.warn('perf observer failed', e); }

  // (2) Passive WebSocket lifecycle watcher (pass-through; observes open/close/error only)
  try {
    const RealWS = window.WebSocket;
    function WrappedWS(url, protocols) {
      const sock = protocols !== undefined ? new RealWS(url, protocols) : new RealWS(url);
      const rec = {
        url: String(url).replace(/wss?:\/\/[^/]+/, '').replace(/https?:\/\/[^/]+/, ''),
        openedAt: Date.now(), open101: false, closedAt: null, code: null, errored: false,
      };
      CAP.ws.push(rec);
      sock.addEventListener('open', () => { rec.open101 = true; });
      sock.addEventListener('close', ev => { rec.closedAt = Date.now(); rec.code = ev.code; });
      sock.addEventListener('error', () => { rec.errored = true; });
      return sock;
    }
    WrappedWS.prototype = RealWS.prototype;
    ['CONNECTING', 'OPEN', 'CLOSING', 'CLOSED'].forEach(k => { WrappedWS[k] = RealWS[k]; });
    window.WebSocket = WrappedWS;
  } catch (e) { console.warn('ws wrap failed', e); }

  // (3) console.error counter (dedup by first 70 chars)
  try {
    const realErr = console.error;
    console.error = function (...a) {
      const k = (a[0] != null ? String(a[0]) : '?').slice(0, 70);
      CAP.consoleErr[k] = (CAP.consoleErr[k] || 0) + 1;
      return realErr.apply(this, a);
    };
  } catch (e) {}

  // mark helper for the cold/warm return test
  window.__mark = label => { CAP.marks.push({ label, at: Date.now() }); console.log('marked:', label); return label; };

  // build + persist + render the report
  window.__report = () => {
    const pct = (a, p) => a.length ? a.slice().sort((x, y) => x - y)[Math.min(a.length - 1, Math.floor(p * a.length))] : null;
    const lat = {};
    for (const k of ['/version', 'selfsubjectrulesreviews', 'clusterversions']) {
      const a = CAP.perf.filter(h => h.path.includes(k)).map(h => h.ms);
      lat[k] = { n: a.length, p50: pct(a, 0.5), p95: pct(a, 0.95), max: a.length ? Math.max(...a) : null, over5s: a.filter(x => x > 5000).length };
    }
    const sar = CAP.perf.filter(h => h.path.includes('selfsubjectrulesreviews')).sort((a, b) => a.start - b.start);
    let overlaps = 0; const gaps = [];
    for (let i = 1; i < sar.length; i++) {
      if (sar[i].cluster === sar[i - 1].cluster) {
        const end = sar[i - 1].start + sar[i - 1].ms;
        if (sar[i].start < end) overlaps++; else gaps.push(sar[i].start - end);
      }
    }
    const out = {
      soakMinutes: Number(((Date.now() - CAP.installedAt) / 60000).toFixed(1)),
      latency_ms: lat,
      sar_same_cluster_overlaps: overlaps,
      sar_gaps_ms: gaps,
      websockets: {
        total: CAP.ws.length,
        opened_101: CAP.ws.filter(w => w.open101).length,
        closed: CAP.ws.filter(w => w.closedAt).length,
        errored: CAP.ws.filter(w => w.errored).length,
        still_open: CAP.ws.filter(w => !w.closedAt).length,
        lifecycles: CAP.ws.map(w => ({ path: w.url, open_ms: w.closedAt ? w.closedAt - w.openedAt : null, close_code: w.code, errored: w.errored, still_open: !w.closedAt })),
      },
      console_errors: CAP.consoleErr,
      marks: CAP.marks,
    };
    try { localStorage.setItem('__CAP_REPORT', JSON.stringify(out)); } catch (e) {}
    let el = document.getElementById('__cap_overlay');
    if (!el) {
      el = document.createElement('div');
      el.id = '__cap_overlay';
      el.style.cssText = 'position:fixed;z-index:2147483647;left:8px;bottom:8px;max-width:560px;max-height:60vh;overflow:auto;background:#111;color:#0f0;font:11px/1.4 monospace;padding:8px;border:1px solid #0f0;white-space:pre-wrap';
      document.body.appendChild(el);
    }
    el.textContent = 'CAP_REPORT\n' + JSON.stringify(out, null, 2);
    console.log('CAP_REPORT', out);
    return out;
  };

  console.log('capture harness installed. Do NOT hard-reload. Use __mark("...") and __report().');
})();
